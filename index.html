// 2. 캘린더 생성 기능 최적화 (로그인 직후 딱 1번만 빌드) React.useEffect(() => { if (!user) return; // React가 메인 레이아웃 DOM을 완전히 그린 후 안전하게 실행되도록 지연 적용 const timer = setTimeout(() => { const calendarEl = document.getElementById('calendar'); if (!calendarEl || calendarInstanceRef.current) return; const calendar = new FullCalendar.Calendar(calendarEl, { // 👇 핵심 수정 1: 첫 화면을 월간 달력(구글 캘린더 형식)으로 변경 initialView: 'dayGridMonth', locale: 'ko', headerToolbar: { left: 'prev,next today', center: 'title', right: 'dayGridMonth,timeGridWeek,timeGridDay' }, slotMinTime: '07:00:00', slotMaxTime: '22:00:00', allDaySlot: false, validRange: { start: todayStr, end: maxDateStr }, selectable: true, select: function(info) { // 👇 핵심 수정 2: 월간 달력에서 날짜 클릭 시 기본 시간(09:00~10:00) 세팅 방어 로직 let startT = info.startStr; let endT = info.endStr; // 클릭한 값이 '2024-05-20' 처럼 날짜만 있는 경우 (월간 뷰) if (startT.length === 10) { startT = `${startT}T09:00`; endT = `${info.startStr}T10:00`; } else { // 주간/일간 뷰에서 시간까지 드래그한 경우 startT = startT.slice(0,16); endT = endT.slice(0,16); } setSelectedSlot({ start: startT, end: endT }); setSelectedEvent(null); setShowNotice(true); }, eventClick: function(info) { const props = info.event.extendedProps; setSelectedEvent(props); setFormData({ id: props.id, dept: props.dept, name: props.name, empNo: props.empNo, startTime: props.startTime, endTime: props.endTime, budget: props.budget, purpose: props.purpose, details: props.details, contact: props.contact }); setShowForm(true); } }); calendar.render(); calendarInstanceRef.current = calendar; }, 150); return () => { clearTimeout(timer); if (calendarInstanceRef.current) { calendarInstanceRef.current.destroy(); calendarInstanceRef.current = null; } }; }, [user]);
